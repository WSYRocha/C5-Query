SELECT DISTINCT P.INDINTEGRAECOMMERCE  AS INTEGRA_ECOMM,
                P.PERSIMILIARECOMMERCE AS SIMILAR_ECOOM,
                P.SEQPRODUTO AS PLU,
                P.DESCCOMPLETA AS PRODUTO,
                C.INDUTILVENDA,
   CASE WHEN EXISTS (SELECT 1 FROM CONSINCO.MAP_AUDITORIA A WHERE A.SEQIDENTIFICA = P.SEQPRODUTO
   AND A.TABELA = 'MAP_PRODUTO'
   AND A.ORIGEM = 'PRODUTO'
   AND A.CAMPO = ' INDINTEGRAECOMMERCE'
   AND NVL(VLRATUAL,'N') = 'N' 
   AND A.USUAUDITORIA != 'AD_WALLACE') THEN 'Inativado' ELSE NULL END IND_INAT
   FROM CONSINCO.MAP_PRODUTO P
INNER JOIN CONSINCO.MAP_PRODCODIGO C
    ON P.SEQPRODUTO = C.SEQPRODUTO
WHERE C.INDUTILVENDA = 'S'
   AND (P.INDINTEGRAECOMMERCE IS NULL OR P.INDINTEGRAECOMMERCE = 'N' OR
       P.PERSIMILIARECOMMERCE IS NULL OR P.PERSIMILIARECOMMERCE = 'N')
   AND C.QTDEMBALAGEM = 1
ORDER BY P.DESCCOMPLETA;
